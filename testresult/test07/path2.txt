int k;
int m;
int i, j, sum;
int f[10];
@(!(m <= k - 2));


@(!(m == k - 2));


i = 0;
@(i <= k - 2);
i = i + 1;
f[i] = 0;
@(i <= k - 2);
i = i + 1;
f[i] = 0;
@(!(i <= k - 2));
f[k - 1] = 1;
i = k;
@(i <= m);
i = i + 1;
sum = 0;
j = i - k;
@(j < i);
j = j + 1;
sum = sum + f[j];
@(j < i);
j = j + 1;
sum = sum + f[j];
@(!(j < i));
f[i] = sum;
@(!(i <= m));
return f[m];

int arr[5];
int left;
int right;
int key;
int hole;
hole = left;
key = arr[left];
@(left < right);
@(left < right && arr[right] >= key);
right = right - 1;
@(left < right && arr[right] >= key);
right = right - 1;
@(!(left < right && arr[right] >= key));

arr[hole] = arr[right];
hole = right;
@(left < right && arr[left] <= key);
left = left + 1;
@(left < right && arr[left] <= key);
left = left + 1;
@(!(left < right && arr[left] <= key));

arr[hole] = arr[left];
hole = left;
@(!(left < right));

arr[hole] = key;
return hole;

int s[10];
int len;
int i, j, gap, temp;
gap = len / 2;
@(gap > 0);
gap = gap / 2;
i = gap;
@(i < len);
i = i + 1;
j = i - gap;
@(j >= 0 && s[j] > s[j + gap]);
j = j - gap;
temp = s[j + gap];
s[j + gap] = s[j];
s[j] = temp;
@(j >= 0 && s[j] > s[j + gap]);
j = j - gap;
temp = s[j + gap];
s[j + gap] = s[j];
s[j] = temp;
@(!(j >= 0 && s[j] > s[j + gap]));
@(!(i < len));
@(!(gap > 0));
return 1;

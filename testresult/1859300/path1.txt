scanf("%d", &k);
ki = k + 1;
printf("%d\n", ki);
@(!(ki % 2 == 0));
int x = (ki + 1) / 2;
printf("%d ", x);
int i = x - 1, j = x + 1;
@(i >= 1 && j <= ki);
i--, j++;
printf("%d %d ", i, j);

@(i >= 1 && j <= ki);
